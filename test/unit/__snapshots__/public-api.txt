defineChannel: {

definePlugin: {
  <const P extends Plugin>(plugin: P) => P
}

defineReadTool: {
  <const S extends ReadParameters, const R extends ResultFields>(tool: PluginReadTool<S, R>) => PluginReadTool<S, R>
}

interface InboundSchema<T> { safeParse(raw: unknown): { success: true; data: T } | { success: false; error: unknown }; }

type ParamsOf<S extends PluginParameters | ReadParameters> = { [K in keyof S as S[K] extends { optional: true } ? never : K]: ValueOf<S[K]>; } & { [K in keyof S as S[K] extends { optional: true } ? K : never]?: ValueOf<S[K]>; };

interface PluginCall<P> { subscriberId: string; params: P; logger: PluginLogger; signal: AbortSignal; }

interface PluginChannel<TInbound = unknown, TOutbound = unknown> { name: string; apiVersion: number; inbound: InboundSchema<TInbound>; parse(request: TInbound): ChannelMessage; render(reply: ChannelReply): TOutbound; push(call: ChannelPush): Promise<void> | void; }

type PluginLogger = {
  error: {
    (message: string, fields?: Record<string, unknown> | undefined) => void
  }
  info: {
    (message: string, fields?: Record<string, unknown> | undefined) => void
  }
  warn: {
    (message: string, fields?: Record<string, unknown> | undefined) => void
  }
}
